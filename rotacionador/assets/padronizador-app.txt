function numberString(value) {
  const string = clean(value).trim();
  if (!/e[+-]?\d/i.test(string)) return string;
  const match = string.match(/^([+-]?)(\d+(?:[.,]\d+)?)[eE]([+-]?\d+)$/);
  if (!match) return string;
  const sign = match[1] === '-' ? '-' : '';
  const mantissa = match[2].replace(',', '.');
  const exponent = Number(match[3]);
  const [whole, fraction = ''] = mantissa.split('.');
  const digits = `${whole}${fraction}`.replace(/^0+(?=\d)/, '') || '0';
  const decimalShift = exponent - fraction.length;
  if (decimalShift >= 0) return sign + digits + '0'.repeat(decimalShift);
  const cutoff = digits.length + decimalShift;
  if (cutoff <= 0) return sign + '0';
  return sign + digits.slice(0, cutoff);
}

function csvHasScientificPhoneData(file, data) {
  if (!/\.csv$/i.test(file.name)) return false;
  const lines = new TextDecoder().decode(data).replace(/^\uFEFF/, '').split(/\r?\n/).filter(Boolean);

  return lines.slice(1).some((line) => scientificNotation(line));
}

function validBrazilianPhone(value) {

function phone(value) {
  let string = numberString(value).replace(/\D/g, '');
  if (!string || /^0+$/.test(string)) return '';
  if (string.length === 10 || string.length === 11) {
    string = '55' + string;
  } else if (string.length === 12 && !string.startsWith('55')) {
    string = '55' + string;
  }
  if (string.length === 14 && string.startsWith('55') && string[4] === '9') {
    string = string.slice(0, 4) + string.slice(5);
  }
  return validBrazilianPhone(string) ? string : '';
}

function documentValue(value, header) {

  const normalizedHeader = key(header);
  if (normalizedHeader.includes('cnpj')) return string.padStart(14, '0');
  if (normalizedHeader.includes('cpf')) return string.padStart(11, '0');
  return string;
}

function personName(value) {
  const string = clean(value).normalize('NFC').replace(/\s+/g, ' ');
  if (!string) return '';
  const lower = new Set(['a', 'as', 'o', 'os', 'e', 'da', 'das', 'de', 'do', 'dos', 'em', 'na', 'nas', 'no', 'nos', 'para', 'por']);
  return string
    .toLocaleLowerCase('pt-BR')
    .split(' ')
    .map((part, index) => {
      if (index && lower.has(part)) return part;
      return part.charAt(0).toLocaleUpperCase('pt-BR') + part.slice(1);
    })
    .join(' ');
}

async function readFile(file) {
  const data = await file.arrayBuffer();
  const workbook = XLSX.read(data, { type: 'array', cellDates: false, raw: false });
  const rows = [];
  for (const sheetName of workbook.SheetNames) {
    const sheet = workbook.Sheets[sheetName];
    const matrix = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: '', raw: false })
      .filter((row) => row.some((value) => clean(value)));
    if (!matrix.length) continue;
    const foneCelularIndex = matrix[0].findIndex((value) => key(value) === 'fonecelular');
    if (foneCelularIndex >= 0) {
      rows.push(...matrix.slice(1)
        .filter((row) => clean(row[foneCelularIndex]))
        .map((row) => ({ Telefone: row[foneCelularIndex], __source: file.name, __sheet: sheetName })));
      continue;
    }
    const width = Math.max(...matrix.map((row) => row.length));
    if (width === 1) {
      const firstCell = clean(matrix[0][0]);
      const hasPhoneHeader = colType(firstCell) === 'phone';
      const values = hasPhoneHeader ? matrix.slice(1) : matrix;
      rows.push(...values
        .filter((row) => clean(row[0]))
        .map((row) => ({ Telefone: row[0], __source: file.name, __sheet: sheetName })));
      continue;
    }
    const list = XLSX.utils.sheet_to_json(sheet, { defval: '', raw: false });
    rows.push(...list.map((row) => ({ ...row, __source: file.name, __sheet: sheetName })));
  }
  if (!rows.length && /\.txt$/i.test(file.name)) {
    const text = new TextDecoder().decode(data);
    rows.push(...text.split(/\r?\n/).filter(Boolean).map((value) => ({ Telefone: value, __source: file.name })));
  }
  return rows;
}

}

    .filter(Boolean);
}

      const before = clean(original[header] ?? '');
      let value = original[header] ?? '';

      if (type === 'phone') {
        phonesByColumn[header] = phoneList(value);
        value = phonesByColumn[header][0] || '';
      } else if (type === 'doc') {
        value = documentValue(value, header);
      } else if (type === 'name') {
        value = personName(value);
      }

      row[header] = value;
      if (type !== 'other' && before !== clean(value)) normalizationStats.changed++;
    }

    const phoneColumns = headers.filter((header) => activeColType(header) === 'phone');
    const expanded = phoneColumns.length ? phoneColumns.flatMap((header) => phonesByColumn[header] || []) : [''];

    if (phoneColumns.length && !expanded.length) {
      normalizationStats.skipped++;
      continue;
    }

    const uniquePhones = [...new Set(expanded)];
    normalizationStats.deduplicated += expanded.length - uniquePhones.length;

    if (phoneColumns.length && uniquePhones.length > 1) {
      normalizationStats.expanded += uniquePhones.length - 1;
      uniquePhones.forEach((value) => {
        const copy = { ...row };
        phoneColumns.forEach((header) => { copy[header] = ''; });
        copy[phoneColumns[0]] = value;
        out.push(copy);
      });
    } else {
      out.push(row);
    }
  }

  const deduplicated = [];
  const seenPhones = new Set();
  const phoneColumns = headers.filter((header) => activeColType(header) === 'phone');
  for (const row of out) {
    const rowPhones = [...new Set(phoneColumns.flatMap((header) => phoneList(row[header] || '')))];
    if (rowPhones.some((value) => seenPhones.has(value))) {
      normalizationStats.deduplicated++;
      continue;
    }
    rowPhones.forEach((value) => seenPhones.add(value));
    deduplicated.push(row);
  }
  return deduplicated;
}

function addControls(rows, includeTestNumbers = PAGE_MODE !== 'zip') {
  const cleanRows = rows.filter((row) => !isControl(row));
  const phoneColumns = headers.filter((header) => activeColType(header) === 'phone');
  const nameColumn = headers.find((header) => activeColType(header) === 'name');
  const docColumn = headers.find((header) => activeColType(header) === 'doc');
  const testNumbers = includeTestNumbers ? testPhoneNumbers() : [];
  if (includeTestNumbers && testPhoneNumbers().length && !phoneColumns.length) {
    throw new Error('Não foi encontrada uma coluna de telefone para inserir os números de teste.');
  }
  const result = [...cleanRows];

  testNumbers.forEach((testNumber) => {
    const row = {};
    headers.forEach((header) => { row[header] = ''; });
    row[phoneColumns[0]] = testNumber;
    result.push(row);
  });

  if (PAGE_MODE === 'default-number') {
    if (!phoneColumns.length) throw new Error('Não foi encontrada uma coluna de telefone para inserir o número padrão.');
    DEFAULT_STANDARD_NUMBERS.forEach((standardNumber, index) => {
      const row = {};
      headers.forEach((header) => { row[header] = ''; });
      if (nameColumn) row[nameColumn] = index ? 'Número padrão 2' : 'Número padrão';
      row[phoneColumns[0]] = standardNumber;
      result.push(row);
    });
    return result;
  }

  CONTROLS.forEach((control, index) => {
    const row = {};
    headers.forEach((header) => { row[header] = ''; });
    if (nameColumn) row[nameColumn] = control.name;
    if (docColumn) row[docColumn] = control.cpf;
    if (phoneColumns[0]) row[phoneColumns[0]] = control.phone;
    const position = Math.min(result.length - 1, Math.max(1, Math.floor(result.length * ((index + 1) / (CONTROLS.length + 1)))));
    result.splice(position + index, 0, row);
  });

  return result;
}

function splitEnabled() {
  return PAGE_MODE === 'zip' && Boolean($('#split-base')?.checked);
}

function splitBaseRows(rows, requestedValue = '') {
  if (rows.length < 2) throw new Error('A base precisa ter pelo menos 2 contatos para ser dividida.');
  const raw = clean(requestedValue);
  const firstCount = raw === '' ? Math.ceil(rows.length / 2) : Number(raw);
  if (!Number.isInteger(firstCount) || firstCount < 1 || firstCount >= rows.length) {
    throw new Error(`Informe de 1 a ${rows.length - 1} contatos no primeiro ZIP.`);
  }
  return [rows.slice(0, firstCount), rows.slice(firstCount)];
}

function currentBaseParts() {
  const customerRows = packageState?.customerRows || mergedRows.filter((row) => !isControl(row));
  if (!splitEnabled()) {
    return [{ number: null, customers: customerRows, rows: mergedRows }];
  }
  return splitBaseRows(customerRows, $('#split-first-count')?.value || '').map((customers, index) => ({
    number: index + 1,
    customers,
    rows: addControls(customers),
  }));
}

function table(rows) {
  const headerRow = headers.map((header, index) => `<th><div class="column-header"><span>${escapeHtml(header)}</span><span class="column-actions"><button class="column-edit" type="button" data-edit-column="${index}" aria-label="Editar título da coluna ${escapeHtml(header)}" title="Editar título">✎</button><button class="column-remove" type="button" data-remove-column="${index}" aria-label="Excluir coluna ${escapeHtml(header)}" title="Excluir esta coluna">×</button></span></div></th>`).join('');
  const body = rows.slice(0, 30).map((row) => `<tr>${headers.map((header) => `<td>${escapeHtml(row[header])}</td>`).join('')}</tr>`).join('');
  return `<table><thead><tr>${headerRow}</tr></thead><tbody>${body}</tbody></table>`;
}

function currentResultRows() {
  return new Set([
    ...mergedRows,
    ...(packageState?.rows || []),
    ...(packageState?.customerRows || []),
    ...resultFiles.flatMap((file) => file.rows || []),
  ]);
}

function refreshAdjustedResult(message) {

  renderZipPreview();
  setPackageReady();
  $('#result-status').textContent = message;
}

function editColumn(index) {
  if (!packageState || index < 0 || index >= headers.length) return;
  const oldHeader = headers[index];
  const input = window.prompt('Novo título da coluna:', oldHeader);
  if (input === null) return;
  const newHeader = clean(input);
  if (!newHeader) {
    status('O título da coluna não pode ficar vazio.');
    return;
  }
  if (newHeader === oldHeader) return;
  if (headers.some((header, currentIndex) => currentIndex !== index && header.toLocaleLowerCase('pt-BR') === newHeader.toLocaleLowerCase('pt-BR'))) {
    status(`Já existe uma coluna chamada “${newHeader}”.`);
    return;
  }

  const preservedType = activeColType(oldHeader);
  currentResultRows().forEach((row) => {
    row[newHeader] = row[oldHeader];
    delete row[oldHeader];
  });
  headers[index] = newHeader;
  headerTypes.delete(oldHeader);
  headerTypes.set(newHeader, preservedType);

  refreshAdjustedResult(`Coluna “${oldHeader}” renomeada para “${newHeader}”. O Excel dentro do ZIP usará o novo título.`);
  status(`Título atualizado para “${newHeader}”.`);
}

function removeColumn(index) {
  if (!packageState || index < 0 || index >= headers.length) return;
  if (headers.length === 1) {
    status('A última coluna não pode ser excluída.');
    $('#result-status').textContent = 'Mantenha pelo menos uma coluna na planilha final.';
    return;
  }

  const [removedHeader] = headers.splice(index, 1);
  currentResultRows().forEach((row) => { delete row[removedHeader]; });
  headerTypes.delete(removedHeader);

  refreshAdjustedResult(`Coluna “${removedHeader}” excluída. O Excel dentro do ZIP será gerado sem ela.`);
  status(`Coluna “${removedHeader}” removida da planilha final.`);
}

function manifestText(rows = mergedRows, partNumber = null, customerCount = Math.max(0, rows.length - CONTROLS.length)) {
  const copyText = clean($('#copy').value);
  const buttonLink = clean($('#button-link').value);
  const creativeLabel = creativeFile ? creativeFile.name : 'Nenhum criativo anexado';
  return [
    'Pacote final gerado pelo padronizador.',
    `Data: ${new Date().toLocaleString('pt-BR')}`,
    `Prompt Master: ${MASTER_PROMPT.version}`,
    partNumber ? `Parte: ${partNumber} de 2` : '',
    `Contatos da base: ${customerCount}`,
    `Linhas finais da base: ${rows.length}`,
    `Copy: ${copyText || 'sem copy informada'}`,

    `Criativo: ${creativeLabel}`,
  ].filter(Boolean).join('\n');
}

function buildPackageDescriptors(rows = mergedRows, partNumber = null, customerCount = Math.max(0, rows.length - CONTROLS.length)) {
  const copyText = clean($('#copy').value);
  const buttonLink = clean($('#button-link').value);
  const baseName = outputFileName('xlsx', partNumber);
  const descriptors = [
    { path: `base/${baseName}`, kind: 'array', data: makeXlsx(rows) },
    { path: 'copy/copy.txt', kind: 'text', data: copyText || 'Sem copy informada.' },
    { path: 'links/link-do-botao.txt', kind: 'text', data: buttonLink || 'Sem link informado.' },
    { path: 'manifesto.txt', kind: 'text', data: manifestText(rows, partNumber, customerCount) },
  ];

  const testNumbers = testPhoneNumbers();
  if (PAGE_MODE === 'zip' && testNumbers.length) {
    descriptors.push({
      path: 'testes/numeros-de-teste.txt',
      kind: 'text',
      data: `${testNumbers.join('\n')}\n`,
    });
  }

  if (creativeFile) {
    descriptors.push({
      path: `criativo/${sanitizeFileName(creativeFile.name)}`,
      kind: 'file',
      data: creativeFile,
    });
  } else {
    descriptors.push({ path: 'criativo/sem-criativo.txt', kind: 'text', data: 'Nenhum criativo foi enviado.' });
  }

  return descriptors;
}

async function buildZipBlob(rows = mergedRows, partNumber = null, customerCount = Math.max(0, rows.length - CONTROLS.length)) {
  const zip = new JSZip();
  for (const item of buildPackageDescriptors(rows, partNumber, customerCount)) {
    zip.file(item.path, item.data);
  }
  return zip.generateAsync({ type: 'blob', compression: 'DEFLATE' });
}

function makeXlsx(rows) {
  const ws = XLSX.utils.json_to_sheet(rows, { header: headers });
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, 'Base');
  return XLSX.write(wb, { bookType: 'xlsx', type: 'array' });
}

async function processFiles() {
  try {
    status('Lendo e normalizando arquivos...');
    const allRows = [];
    for (const file of selectedFiles) {
      allRows.push(...await readFile(file));
    }

    const union = [];
    allRows.forEach((row) => {
      Object.keys(row).forEach((header) => {
        if (!header.startsWith('__') && !union.includes(header)) union.push(header);
      });
    });

    headers = ensureControlColumns(union);
    headerTypes = new Map(headers.map((header) => [header, colType(header)]));
    const normalizedRows = normalizeRows(allRows);
    const customerRows = normalizedRows.filter((row) => !isControl(row));
    mergedRows = addControls(customerRows);
    resultFiles = [{ name: outputFileName('xlsx'), rows: mergedRows }];
    packageState = {
      rows: mergedRows,
      customerRows,
      copyText: clean($('#copy').value),
      buttonLink: clean($('#button-link').value),
      creativeFile,
    };

    const testCount = testPhoneNumbers().length;
    const real = PAGE_MODE === 'default-number' ? mergedRows.length : customerRows.length;
    $('#metrics').innerHTML = [
      ['Arquivos recebidos', selectedFiles.length],
      ['Colunas ajustáveis', normalizationStats.recognized],
      ['Campos alterados', normalizationStats.changed],
      ['Disparos previstos', real],
      ['Telefones repetidos', normalizationStats.deduplicated],
      ['Linhas ignoradas', normalizationStats.skipped],
      ...(PAGE_MODE === 'default-number' ? [['Números de teste', testCount]] : []),
      [PAGE_MODE === 'default-number' ? 'Números padrão' : 'Controles aplicados', PAGE_MODE === 'default-number' ? DEFAULT_STANDARD_NUMBERS.length : CONTROLS.length],
    ].map(([label, value]) => `<div class="metric"><span>${label}</span><strong>${value}</strong></div>`).join('');
    $('#applied').textContent = PAGE_MODE === 'default-number'
      ? `Números padrão ${DEFAULT_STANDARD_NUMBERS.join(' e ')} incluídos na planilha. ${testCount} número(s) de cliente entraram como teste direto na planilha.`
      : `Prompt Master ${MASTER_PROMPT.version} aplicado: ${normalizationStats.changed} campo(s) normalizado(s), ${normalizationStats.expanded} telefone(s) separado(s), ${normalizationStats.deduplicated} telefone(s) repetido(s) desconsiderado(s), ${normalizationStats.skipped} linha(s) inválida(s) ou sem telefone ignorada(s) e ${CONTROLS.length} controles incluídos${splitEnabled() ? ' em cada parte' : ''}.`;
    $('#preview').innerHTML = table(mergedRows);
    renderSplitControls();
    renderZipPreview();
    $('#result').classList.remove('hidden');
    $('#result-status').textContent = PAGE_MODE === 'zip'
      ? splitEnabled() ? 'Base ajustada pronta. Os 2 ZIPs já podem ser baixados.' : 'Base ajustada pronta. O ZIP já pode ser baixado.'
      : PAGE_MODE === 'default-number' ? 'Prévia pronta. Edite as colunas se precisar e baixe a planilha pronta.' : 'Base ajustada pronta. Baixe apenas a planilha.';
    setPackageReady();
    configureMode();
    status('Concluído.');
  } catch (error) {
    resetPackageState();
    status(`Falha: ${error?.message || error}`);
  }
}

  if (!packageState) return;
  download(resultFiles[0].name, new Blob([makeXlsx(packageState.rows)], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' }));
};

$('#download-zip').onclick = async () => {
  if (!packageState || PAGE_MODE !== 'zip') return;
  try {
    $('#download-zip').disabled = true;
    const parts = currentBaseParts();
    $('#result-status').textContent = parts.length === 2 ? 'Montando os 2 ZIPs...' : 'Montando ZIP...';
    const downloads = await Promise.all(parts.map(async (part) => ({
      name: outputFileName('zip', part.number),
      blob: await buildZipBlob(part.rows, part.number, part.customers.length),
    })));
    downloads.forEach((item) => download(item.name, item.blob));
    $('#result-status').textContent = parts.length === 2
      ? '2 ZIPs prontos para download. Se o navegador perguntar, permita vários downloads.'
      : 'ZIP pronto para download.';
  } catch (error) {
    $('#result-status').textContent = `Falha ao gerar ZIP: ${error?.message || error}`;
  } finally {
    setPackageReady();
  }
};
